<!DOCTYPE html>
<html>
<head>
<meta charset="UTF-8">
<title>Corrida do Estudo</title>
<style>
  body { margin: 0; overflow: hidden; background: #87CEEB; }
  canvas { display: block; margin: auto; }
</style>
</head>
<body>

<canvas id="game" width="400" height="600"></canvas>

<script>
const canvas = document.getElementById("game");
const ctx = canvas.getContext("2d");

// ===== IMAGENS =====
const playerImg = new Image();
playerImg.src = "https://i.imgur.com/7QKQZ8G.png";

const enemyImg = new Image();
enemyImg.src = "https://i.imgur.com/Z6XbKQp.png";

const roadImg = new Image();
roadImg.src = "https://i.imgur.com/9j8KQpP.png";

// ===== SOM =====
let music = new Audio("https://www.soundhelix.com/examples/mp3/SoundHelix-Song-1.mp3");
music.loop = true;
music.volume = 0.3;

let crashSound = new Audio("https://www.soundjay.com/mechanical/explosion-01.mp3");

// ===== ESTADOS =====
let state = "menu";

// ===== PLAYER =====
let player = { x: 170, y: 500, w: 50, h: 80 };

// ===== GAME =====
let enemies = [];
let score = 0;
let highScore = localStorage.getItem("recorde") || 0;
let bestRank = localStorage.getItem("bestRank") || "Iniciante 🟤";

let speed = 4;
let roadY = 0;
let gameOver = false;

// ===== SISTEMA DE RANK =====
function getRank(score) {
  if (score < 50) return "Iniciante 🟤";
  if (score < 150) return "Bronze 🟫";
  if (score < 300) return "Prata ⚪";
  if (score < 500) return "Ouro 🟡";
  if (score < 800) return "Diamante 🔷";
  return "Lendário 🔥";
}

// ===== RESET =====
function resetGame() {
  enemies = [];
  score = 0;
  speed = 4;
  gameOver = false;
  player.x = 170;
}

// ===== CONTROLES =====
canvas.addEventListener("mousemove", e => {
  player.x = e.clientX - canvas.offsetLeft - player.w/2;
});

canvas.addEventListener("touchmove", e => {
  player.x = e.touches[0].clientX - canvas.offsetLeft - player.w/2;
});

// ===== CLIQUE =====
canvas.addEventListener("click", (e) => {
  let mouseX = e.clientX - canvas.offsetLeft;
  let mouseY = e.clientY - canvas.offsetTop;

  // iniciar jogo
  if (state === "menu") {
    state = "game";
    music.play();
    return;
  }

  // botão reiniciar
  if (gameOver) {
    if (
      mouseX > canvas.width/2 - 80 &&
      mouseX < canvas.width/2 + 80 &&
      mouseY > 300 &&
      mouseY < 350
    ) {
      resetGame();
    }
  }
});

// ===== INIMIGOS =====
function spawnEnemy() {
  let x = Math.random() * (canvas.width - 50);
  enemies.push({ x: x, y: -100, w: 50, h: 80, passed: false });
}

// ===== UPDATE =====
function update() {
  if (state !== "game" || gameOver) return;

  roadY += speed;

  enemies.forEach(e => {
    e.y += speed;

    if (!e.passed && e.y > player.y) {
      score += 3;
      e.passed = true;
    }

    if (
      player.x < e.x + e.w &&
      player.x + player.w > e.x &&
      player.y < e.y + e.h &&
      player.y + player.h > e.y
    ) {
      crashSound.play();
      gameOver = true;

      let currentRank = getRank(score);

      if (score > highScore) {
        localStorage.setItem("recorde", score);
      }

      localStorage.setItem("bestRank", currentRank);
    }
  });

  enemies = enemies.filter(e => e.y < canvas.height);

  speed = 4 + Math.floor(score / 50);
}

// ===== DESENHO =====
function draw() {
  ctx.clearRect(0,0,canvas.width,canvas.height);

  // fundo
  ctx.fillStyle = "#87CEEB";
  ctx.fillRect(0,0,canvas.width,canvas.height);

  // estrada
  ctx.drawImage(roadImg, 0, roadY % 600 - 600, 400, 600);
  ctx.drawImage(roadImg, 0, roadY % 600, 400, 600);

  // player
  ctx.drawImage(playerImg, player.x, player.y, player.w, player.h);

  // inimigos
  enemies.forEach(e => {
    ctx.drawImage(enemyImg, e.x, e.y, e.w, e.h);
  });

  // UI
  ctx.fillStyle = "black";
  ctx.font = "20px Arial";
  ctx.fillText("Score: " + score, 10, 30);
  ctx.fillText("Recorde: " + highScore, 10, 60);

  let rank = getRank(score);
  ctx.fillText("Rank: " + rank, 10, 90);
  ctx.fillText("Melhor Rank: " + bestRank, 10, 120);

  // MENU
  if (state === "menu") {
    ctx.textAlign = "center";
    ctx.font = "32px Arial";
    ctx.fillText("Corrida do Estudo", canvas.width / 2, 250);

    ctx.font = "18px Arial";
    ctx.fillText("Desvie dos carros e evolua de rank!", canvas.width / 2, 280);

    ctx.fillText("Clique para Jogar", canvas.width / 2, 320);
  }

  // GAME OVER + botão
  if (gameOver) {
    ctx.textAlign = "center";

    ctx.fillStyle = "red";
    ctx.font = "30px Arial";
    ctx.fillText("GAME OVER", canvas.width / 2, 260);

    // botão
    ctx.fillStyle = "black";
    ctx.fillRect(canvas.width/2 - 80, 300, 160, 50);

    ctx.fillStyle = "white";
    ctx.font = "20px Arial";
    ctx.fillText("Reiniciar", canvas.width / 2, 335);
  }
}

// ===== LOOP =====
function loop() {
  update();
  draw();
  requestAnimationFrame(loop);
}

// spawn
setInterval(spawnEnemy, 1200);

loop();
</script>

</body>
</html>
